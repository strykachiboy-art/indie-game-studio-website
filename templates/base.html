// js/main.js document.addEventListener('DOMContentLoaded', function () { /* ---------- Nav drawer (nav.html) ---------- */ const toggle = document.getElementById('nav-toggle'); const drawer = document.getElementById('nav-drawer'); const backdrop = document.getElementById('nav-backdrop'); if (toggle && drawer && backdrop) { const bars = toggle.querySelectorAll('.bar'); const drawerLinks = drawer.querySelectorAll('.drawer-link'); let lastFocused = null; function openDrawer() { lastFocused = document.activeElement; toggle.setAttribute('aria-expanded', 'true'); drawer.setAttribute('aria-hidden', 'false'); drawer.classList.remove('translate-x-full'); backdrop.classList.remove('hidden'); requestAnimationFrame(() => backdrop.classList.remove('opacity-0')); document.body.style.overflow = 'hidden'; bars[0].style.transform = 'translateY(7px) rotate(45deg)'; bars[1].style.opacity = '0'; bars[2].style.transform = 'translateY(-7px) rotate(-45deg)'; drawerLinks.forEach((el, i) => { el.style.transitionDelay = (i * 40) + 'ms'; el.classList.add('transition-opacity'); requestAnimationFrame(() => el.classList.remove('opacity-0')); }); const first = drawer.querySelector('a, button'); if (first) first.focus(); } function closeDrawer() { toggle.setAttribute('aria-expanded', 'false'); drawer.setAttribute('aria-hidden', 'true'); drawer.classList.add('translate-x-full'); backdrop.classList.add('opacity-0'); document.body.style.overflow = ''; bars[0].style.transform = ''; bars[1].style.opacity = ''; bars[2].style.transform = ''; drawerLinks.forEach(el => el.classList.add('opacity-0')); setTimeout(() => backdrop.classList.add('hidden'), 300); if (lastFocused) lastFocused.focus(); } toggle.addEventListener('click', () => { toggle.getAttribute('aria-expanded') === 'true' ? closeDrawer() : openDrawer(); }); backdrop.addEventListener('click', closeDrawer); document.addEventListener('keydown', (e) => { if (e.key === 'Escape' && toggle.getAttribute('aria-expanded') === 'true') closeDrawer(); }); drawer.addEventListener('keydown', (e) => { if (e.key !== 'Tab') return; const list = Array.from(drawer.querySelectorAll('a, button')); const first = list[0], last = list[list.length - 1]; if (e.shiftKey && document.activeElement === first) { e.preventDefault(); last.focus(); } else if (!e.shiftKey && document.activeElement === last) { e.preventDefault(); first.focus(); } }); } /* ---------- Password visibility toggles (login/register) ---------- */ document.querySelectorAll('[data-pw-toggle]').forEach(btn => { btn.addEventListener('click', () => { const field = btn.previousElementSibling; field.type = field.type === 'password' ? 'text' : 'password'; }); }); /* ---------- Password strength meter (register) ---------- */ const pwField = document.querySelector('[data-pw-strength]'); if (pwField) { const bars = [1, 2, 3, 4].map(n => document.getElementById(`pw-bar-${n}`)); const label = document.getElementById('pw-label'); const colors = { weak: '#f87171', okay: '#facc15', good: '#4ade80', strong: 'var(--color-accent, #4ade80)' }; pwField.addEventListener('input', () => { const value = pwField.value; bars.forEach(b => b.style.backgroundColor = ''); if (!value) { label.textContent = '\u00A0'; return; } let score = 0; if (value.length >= 8) score++; if (/[A-Z]/.test(value) && /[a-z]/.test(value)) score++; if (/\d/.test(value)) score++; if (/[^A-Za-z0-9]/.test(value)) score++; const levels = [ { key: 'weak', text: 'Weak' }, { key: 'okay', text: 'Okay' }, { key: 'good', text: 'Good' }, { key: 'strong', text: 'Strong' }, ]; const level = levels[Math.max(0, score - 1)]; for (let i = 0; i < score; i++) bars[i].style.backgroundColor = colors[level.key]; label.textContent = level.text; }); } /* ---------- Flash message dismiss / auto-dismiss ---------- */ document.querySelectorAll('[data-flash]').forEach(el => { const btn = el.querySelector('[data-flash-dismiss]'); if (btn) { btn.addEventListener('click', () => { el.style.opacity = '0'; setTimeout(() => el.remove(), 300); }); } if (el.dataset.autodismiss === 'true') { setTimeout(() => { el.style.opacity = '0'; setTimeout(() => el.remove(), 300); }, 4000); } }); /* ---------- Scroll reveal (homepage, about, etc.) ---------- */ const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches; const revealTargets = document.querySelectorAll('.reveal'); if (revealTargets.length && !prefersReducedMotion && 'IntersectionObserver' in window) { revealTargets.forEach(el => el.classList.add('opacity-0', 'translate-y-3', 'transition-all', 'duration-700')); const io = new IntersectionObserver((entries) => { entries.forEach(entry => { if (entry.isIntersecting) { entry.target.classList.remove('opacity-0', 'translate-y-3'); io.unobserve(entry.target); } }); }, { threshold: 0.15 }); revealTargets.forEach(el => io.observe(el)); } });
